```python
import json
from typing import Dict, Any, Tuple

# Design System V2.0의 핵심 규칙을 하드코딩 (실제로는 config 파일에서 로드해야 함)
DESIGN_RULES = {
    "Pain": {"color": "#FFD700", "text_style": "Montserrat Bold", "icon": "⚠️"},
    "Gain": {"color": "#FFFFFF", "text_style": "Noto Sans KR", "icon": "✅"},
    "Neutral": {"color": "#F5F5F5", "text_style": "Noto Sans KR", "icon": "📊"}
}

class DesignValidatorService:
    """
    System Stability Report의 데이터 값과 비교하여, 마케팅 자산이 적절한 디자인 컨텍스트를 사용했는지 검증합니다.
    [SRP] 오직 디자인 규칙의 논리적 일관성을 체크하는 역할만 수행합니다.
    """

    def __init__(self, design_rules: Dict[str, Any]):
        self.design_rules = design_rules

    def _get_context(self, kpi_value: float, threshold: float) -> str:
        """KPI 값에 따라 컨텍스트('Pain', 'Gain', 'Neutral')를 결정합니다."""
        if kpi_value < threshold * 0.8: # 임계치 대비 20% 미만 = Pain
            return "Pain"
        elif kpi_value >= threshold * 1.5: # 임계치 대비 150% 이상 = Gain
            return "Gain"
        else:
            return "Neutral"

    def check_design_compliance(self, report_data: Dict[str, Any], kpi_key: str, kpi_threshold: float) -> Tuple[bool, str]:
        """
        주어진 KPI 값에 기반하여, 해당 자산이 적절한 디자인 컨텍스트를 사용했는지 검증합니다.

        Args:
            report_data: System Stability Report에서 추출된 데이터 딕셔너리.
            kpi_key: 검증할 KPI의 키 (예: 'Engagement').
            kpi_threshold: 이 KPI가 안정적이라고 판단하는 기준 임계치 값.

        Returns:
            Tuple[bool, str]: 검증 통과 여부와 상세 메시지.
        """
        if kpi_key not in report_data or not isinstance(report_data[kpi_key], (int, float)):
            return False, f"KPI '{kpi_key}' 데이터가 보고서에 존재하지 않거나 유효한 숫자 형식이 아닙니다."

        actual_value = float(report_data[kpi_key])
        context = self._get_context(actual_value, kpi_threshold)
        
        # 예시 로직: 만약 컨텍스트가 'Pain'인데, 자산의 주장하는 바가 긍정적이라면 실패 처리.
        if context == "Pain":
            is_asset_positive = ("최종 점수가 높습니다" in str(report_data) or "성공" in str(report_data))
            if is_asset_positive:
                return False, f"[디자인 오류] '{kpi_key}'는 현재 Pain 컨텍스트({DESIGN_RULES['Pain']['icon']})에 해당합니다. 자산의 긍정적 메시지('성공')와 논리적으로 모순됩니다."
        
        # 추가 검증 로직 (예: Growth가 낮으면, 'Gap Score' 시각화는 반드시 빨간 경고색을 써야 한다.)
        if kpi_key == "Growth" and context == "Pain":
            return True, f"[디자인 OK] '{kpi_key}'의 Pain 컨텍스트({DESIGN_RULES['Pain']['icon']})가 적절히 반영되었습니다. (실제 자산에서 {DESIGN_RULES['Pain']['color']} 계열 색상을 사용했는지 확인하세요.)"
        
        return True, f"[디자인 OK] '{kpi_key}'의 컨텍스트({context})는 디자인과 일치하는 것으로 보입니다."

if __name__ == '__main__':
    # 테스트 케이스: Growth가 낮아 Pain일 때 검증
    validator = DesignValidatorService(DESIGN_RULES)
    test_report = {"Growth": 15, "Engagement": 80} # 임계치 가정 (Growth < 30*0.8=24)
    success, message = validator.check_design_compliance(test_report, "Growth", kpi_threshold=30)
    print(f"테스트 결과: 성공={success}, 메시지={message}")
```

import json
from typing import Dict, Any

# ========================================================
# 1. 비즈니스 상수 (Conversion Factors) 정의 - [근거: CEO 지시]
# 이 값들은 나중에 DB 또는 별도의 Config API에서 로드되어야 합니다.

# KPI별 재정적 가치 변환 계수 (예: 1점당 기대 매출/유지 비용 절감액)
KPI_CONVERSION_FACTORS = {
    "growth": 50,     # Growth Score 1점당 예상 가치 ($50)
    "engagement": 20, # Engagement Score 1점당 유지 가치 ($20)
    "monetization": 80 # Monetization Score 1점당 전환 기여 가치 ($80)
}

def calculate_roi(diagnosis_data: Dict[str, Any]) -> float:
    """
    진단 데이터(DiagnosisScore JSON 구조 예상)를 입력받아 ROI 점수를 계산합니다.

    Args:
        diagnosis_data: API 응답 형태의 진단 결과 딕셔너리.
                         필수 키: 'kpi_metrics' (Dict[str, float])

    Returns:
        총 ROI 점수 (float). 계산 실패 시 0.0을 반환합니다.
    """
    if not diagnosis_data or "kpi_metrics" not in diagnosis_data:
        print("⚠️ 경고: 유효한 KPI 메트릭이 포함되지 않은 진단 데이터를 받았습니다.")
        return 0.0

    kpis = diagnosis_data["kpi_metrics"]
    total_roi = 0.0
    
    # 각 KPI별로 가중치 적용하여 ROI 점수 계산
    try:
        for kpi_name, score in kpis.items():
            if kpi_name not in KPI_CONVERSION_FACTORS:
                print(f"⚠️ 경고: 알 수 없는 KPI '{kpi_name}'가 발견되어 무시합니다.")
                continue
            
            # ROI = 점수 * 변환 계수
            roi_contribution = score * KPI_CONVERSION_FACTORS[kpi_name]
            total_roi += roi_contribution

    except TypeError as e:
        print(f"❌ 치명적 에러 발생: 데이터 타입 오류. 입력 데이터를 확인하세요. ({e})")
        return 0.0
    
    # 최종 로직 검증: ROI는 점수가 높을수록 높아져야 합니다. (단순 합산이므로 OK)
    return round(total_roi, 2)


def generate_detailed_report(diagnosis_data: Dict[str, Any], user_context: Dict[str, str]) -> Dict[str, Any]:
    """
    최종 보고서 구조를 생성하고 ROI 계산 결과를 포함합니다.
    
    Args:
        diagnosis_data: 진단 결과 데이터 딕셔너리.
        user_context: 사용자 컨텍스트 (예: 'role', 'subscription_level').

    Returns:
        완성된 상세 보고서 구조 딕셔너리.
    """
    # ROI 계산을 통해 핵심 지표를 도출합니다.
    roi_score = calculate_roi(diagnosis_data)
    
    report = {
        "summary": "AI 보컬 성장 종합 진단 리포트",
        "diagnostics": diagnosis_data,
        "financial_assessment": {
            "calculated_roi_score": roi_score,
            "interpretation": f"{'매우 높은 가치': 'ROI가 높음', '보통 수준': '적절한 관리가 필요'} 등 (추후 상세 로직 추가)",
            "recommended_action": "KPI 개선을 위한 맞춤형 학습 모듈 추천" # Placeholder
        },
        "user_context_applied": user_context
    }
    return report

### 2. 핵심 계산 로직 정의 (Calculation Logic)

단순 합산이 아닌, **가중치와 비즈니스 요소를 결합**하여 최종 ROI를 산출합니다.

#### A. 기본 ROI 스코어 계산 (Technical Value)
$$ \text{Base\_ROI} = (\text{KPI\_GROWTH\_SCORE} \times W_G) + (\text{KPI\_ENGAGEMENT\_SCORE} \times W_E) $$
*   $W_G$: Growth 가중치. **(Weight of Growth)**
*   $W_E$: Engagement 가중치. **(Weight of Engagement)**

#### B. 재무적 ROI 산출 (Monetary Value)
Base\_ROI를 실제 돈으로 환산합니다. 이 단계에서 시장 가격대와 LTV가 개입됩니다.

$$ \text{Final\_ROI} = (\text{Base\_ROI} \times C_{Factor}) \times M_{\text{Tier}} $$

*   $C_{Factor}$: KPI 변환 계수 (Conversion Factor). **(예: $50, $20)**
    *   **[필요 논리]:** 이 값은 *'만약 이 점수를 개선한다면, 우리 서비스가 절감할 수 있는 비용'*을 반영해야 합니다. (ex: Growth Score 1점 = 코치 시간 절약분 $X)
*   $M_{\text{Tier}}$: 티어별 매출 증대 승수 (Monetization Multiplier). **(Critical Factor)**
    *   **[필요 논리]:** 이 점수는 현재 구독 플랜 대비 '다음 단계의 서비스'가 필요한 정도를 나타냅니다. $M_{Tier}$는 사용자에게 다음 단계를 구매하도록 유도하는 장치입니다.

def run_full_diagnostic_validation(diagnosis_context: Dict[str, Any], diagnosis_data: Dict[str, Any]) -> Dict[str, Any]:
    """
