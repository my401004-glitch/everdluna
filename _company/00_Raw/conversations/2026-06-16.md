우리의 목표는 단순한 매출 보고가 아닌, **'AI 교육 플랫폼이 사용자에게 제공하는 실제 경제적 가치'**를 측정하는 것입니다. 따라서 다음의 5가지 핵심 변수를 중심으로 KPI 및 ROI 시뮬레이션 모델을 설계해야 합니다.

| 구분 | 핵심 재무 변수 (KPI) | 정의 및 계산 로직 (전략적 관점) | 연결되는 시스템 데이터 |
| :--- | :--- | :--- | :--- |
| **1. 수익 (Revenue)** | **MRR/ARR** | 월간/연간 반복 매출. 구독 모델의 핵심 지표. | 결제 기록, 구독 상태 |
| **2. 비용 (Cost)** | **CAC (Customer Acquisition Cost)** | 신규 고객 1명을 확보하는 데 드는 총 마케팅 및 운영 비용. | 마케팅 비용, 광고 집행액, 인건비 등 |
| **3. 가치 (Value)** | **LTV (Customer Lifetime Value)** | 한 명의 고객이 플랫폼에 기여하는 총 예상 수익. (MRR / 예상 이탈률) | 구독 기간, 평균 매출 |
| **4. 효율성 (Efficiency)** | **LTV:CAC Ratio** | 고객 생애 가치 대비 획득 비용 비율. (최소 3:1 목표 설정) | 변수 2와 3 |
| **5. 유지율 (Retention)** | **Churn Rate (이탈률)** | 특정 기간 동안 구독을 취소하거나 이탈한 고객의 비율. (핵심 성장 지표) | 구독 해지 기록, 활성 사용자 수 |

**📌 전략적 결론:** 우리는 단순한 매출 증대를 넘어, **LTV/CAC 비율을 극대화하고 Churn Rate를 최소화**하는 것을 최우선 목표로 설정해야 합니다. AI의 역할은 이 변수들을 실시간으로 측정하고 시각화하여 사용자에게 '성장 서사'를 제공하는 데 집중되어야 합니다.

/**
 * @description 내부 계산 함수 (Pure Function으로 설계하여 테스트 용이성 확보)
 */
const calculateFinancialMetrics = (marketing: any[], subscriptions: any[]): FinancialImpactData => {
    // --- 1. LTV (Lifetime Value) 계산 ---
    // LTV = (평균 월 매출 / 예상 이탈률) * 총 구독 기간 등 복잡한 공식 사용
    const avgMonthlyRevenue = calculateAverageMRR(subscriptions);
    const churnRate = calculateChurnRate(subscriptions); // Churn Rate를 먼저 구함
    const ltv: number = Math.round((avgMonthlyRevenue / churnRate) * 100) / 100;

    // --- 2. CAC (Customer Acquisition Cost) 계산 ---
    // 총 마케팅 비용 합계 / 유입된 사용자 수
    const totalCost = marketing.reduce((sum: number, record: any) => sum + parseFloat(record.cost), 0);
    const uniqueUsersAcquired = new Set(marketing.map(r => r.user_id)).size;
    const cac: number = uniqueUsersAcquired > 0 ? totalCost / uniqueUsersAcquired : 0;

    // --- 3. LTV:CAC Ratio 계산 ---
    const ltvToCacRatio: number = cac > 0 ? Math.round(ltv / cac * 100) / 100 : Infinity;


    return {
        // 시뮬레이션 결과 구조를 정의합니다. (프론트엔드 스키마와 일치해야 함)
        simulationDate: new Date().toISOString(),
        kpis: {
            ltv_estimate: parseFloat(ltv.toFixed(2)), // 예상 고객 생애 가치
            cac_cost: parseFloat(cac.toFixed(2)),   // 고객 획득 비용
            churn_rate: parseFloat(calculateChurnRate(subscriptions).toFixed(4)), // 이탈률 (소수점 4자리)
            ltv_to_cac_ratio: ltvToCacRatio,          // LTV/CAC 비율
        },
        risk_assessment: generateRiskAssessment(ltvToCacRatio, churnRate), // 리스크 평가 (문자열)
        opportunity_score: calculateOpportunityScore(avgMonthlyRevenue, ltvToCacRatio) // 기회 점수 (0-100점)
    };
};

        { campaignId: 'C001', type: 'Google Ads', spend: 500, reach: 3000, conversionRate: 0.02 }, // 초기 리스크 구간 (비효율적 광고)
        { campaignId: 'C002', type: 'Instagram Reels', spend: 800, reach: 12000, conversionRate: 0.04 }, // 개선된 활동 (고효율)

        { period: 'Q2', count: 15, averageRevenue: 4500 } // 성장의 지표

    // --- 💰 핵심 로직 구현 시작 ---

    // 1. 리스크(Risk) 측정: 비효율적 마케팅 활동 및 낮은 전환율 기반 비용 증가 예측
    let totalCostOverrun = 0;
    let highRiskScore = 0;

    mockMarketingActivities.forEach((activity, index) => {
        if (index === 0 && activity.conversionRate < 0.025) { // 첫 번째 활동(가장 낮은 효율성 가정)을 리스크로 간주
            const costOverrun = activity.spend * 0.1; // 예: 비용의 10%가 손실될 가능성
            totalCostOverrun += costOverrun;
            highRiskScore += Math.min(1, (0.025 - activity.conversionRate) / 0.02);
        }
    });

    // 2. 기회(Opportunity) 측정: 높은 효율성과 구독 성장 기반 예상 수익 증가 예측
    const currentRevenue = mockSubscriptionHistory[mockSubscriptionHistory.length - 1].averageRevenue;
    let projectedGrowthAmount = 0;

    }

    // 3. 최종 지표 계산
    const netImpactValue = projectedGrowthAmount - totalCostOverrun;
    const financialRiskScore = Math.min(10, highRiskScore * 10); // 점수 스케일 조정 (0-10)

    // --- 📈 결과 구조화 및 반환 ---
    return {
        userId: userId,
        timestamp: new Date().toISOString(),
        financialImpactSummary: {
            risk_description: `현재 마케팅 활동의 비효율성(최대 ${Math.round(totalCostOverrun)} 예상 손실)으로 인한 잠재적 리스크가 존재합니다.`,
            opportunity_description: `고효율 캠페인과 구독 성장을 기반으로 다음 분기 약 ${Math.round(projectedGrowthAmount - currentRevenue)}원의 추가 수익 기회가 포착됩니다.`,
            net_impact_value: parseFloat(netImpactValue.toFixed(2)), // 최종 순수 이익 기여도
            financial_risk_score: Math.round(financialRiskScore), // 0~10점 (높을수록 리스크)
        },
        // 추가적인 세부 지표를 포함하여 프론트엔드에 풍부한 컨텍스트 제공

            projectedRevenueIncrease: parseFloat((projectedGrowthAmount - currentRevenue).toFixed(2)),
            bestPerformingCampaignId: mockMarketingActivities.find(a => a.conversionRate === 0.10)?.campaignId || 'N/A',
        }
    };
}
